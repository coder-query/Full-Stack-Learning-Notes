# 五、linux基础命令篇

## 1，命令行解释

```bash
[root@Kylin-oldboy ~]# 

root            #当前登录的用户【皇帝用户】
Kylin-oldboy    #主机名称【企业当中，都是以业务为单位设置名称】
~               #当前所在的路径，当前用户的家目录/root
				#/home/username/
#               #root用户显示的是#，普通用户现实的是$

[root@Kylin-oldboy ~]# useradd oldboy
[root@Kylin-oldboy ~]# su - oldboy 
[oldboy@Kylin-oldboy ~]$ exit
注销

#ubuntu
root@oldboy:~#
```

## 4，关机重启命令

```bash
#关机
	shutdown            #关机
	shutdown  -h now/5  #现在或者5分钟后关机
	shutdown  -c        #取消关机、取消重启    
	halt                #立刻关机
	poweroff            #立刻关机
	init 0              #立刻关机

#重启
	reboot              #立刻重启
	shutdown  -r now/5  #现在或者5分钟后重启
	init 6              #立刻重启
```
