### · 网卡配置文件

#### 1，centos/kylin网卡配置文件

```bash
[root@K-oldboy26 ~]# vim /etc/sysconfig/network-scripts/ifcfg-ens33 
TYPE=Ethernet           #以太网类型的网络
PROXY_METHOD=none
BROWSER_ONLY=no
BOOTPROTO=none          #手动获取ip还是自动获取ip？？【手动：static/none】【自动：DHCP】
DEFROUTE=yes            #允许将本机网卡当作路由器使用；
IPV4_FAILURE_FATAL=yes  
IPV6INIT=yes
IPV6_AUTOCONF=yes
IPV6_DEFROUTE=yes
IPV6_FAILURE_FATAL=no
IPV6_ADDR_GEN_MODE=stable-privacy
NAME=ens33              #软件网卡名称
UUID=2e6d0bbc-6e71-4c93-8b47-bc73d4f1bc0c   #mac地址【全球唯一】
DEVICE=ens33            #硬件网卡名
ONBOOT=yes              #网卡是否开机自动运行
IPADDR=09-权限管理.0.0.200       #ip地址
PREFIX=24               #子网掩码
GATEWAY=09-权限管理.0.0.2        #网关地址
DNS1=223.5.5.5          #DNS地址（优先级第二）
IPV6_PRIVACY=no

- kylin
[root@K-oldboy26 ~]# ifdown ens33 && ifup ens33
-centos
[root@oldboy26 ~]# systemctl restart network
```
